<script>
(function(){
  const bubblesGroup = document.getElementById('bubbles');
  const fish1 = document.getElementById('fish1');
  const fish2 = document.getElementById('fish2');

  const fishState = [
    { el: fish1, x: 200, y: 300, speed: 50, dir: 1 },
    { el: fish2, x: 700, y: 150, speed: 110, dir: -1 }
  ];

  const bubbles = [];
  const numBubbles = 9;
  const seafloorY = 600;

  function createBubble(initial) {
    const r = 3 + Math.random() * 7;
    const speed = 30 + (10 - r) * 6;
    const x = 50 + Math.random() * 900;
    const y = initial ? seafloorY + Math.random() * 80 : -r - Math.random() * 100;
    return { x, y, r, speed, wobble: Math.random() * Math.PI * 2, wobbleSpeed: 1 + Math.random() };
  }

  for (let i = 0; i < numBubbles; i++) {
    const b = createBubble(true);
    const circle = document.createElementNS('http://www.w3.org/2000/svg', 'circle');
    circle.setAttribute('fill', 'rgba(200,240,255,0.5)');
    circle.setAttribute('stroke', 'rgba(255,255,255,0.8)');
    circle.setAttribute('stroke-width', '1');
    bubblesGroup.appendChild(circle);
    b.el = circle;
    bubbles.push(b);
  }

  let lastTime = performance.now();

  function animate(now) {
    const dt = Math.min((now - lastTime) / 1000, 0.1);
    lastTime = now;

    fishState.forEach(f => {
      f.x += f.speed * f.dir * dt;
      const off = 150;
      if (f.dir > 0 && f.x > 1000 + off) {
        f.x = -off;
      } else if (f.dir < 0 && f.x < -off) {
        f.x = 1000 + off;
      }
      if (f.el === fish1) {
        f.el.setAttribute('transform', `translate(${f.x},${f.y})`);
      } else {
        f.el.setAttribute('transform', `translate(${f.x},${f.y}) scale(-1,1)`);
      }
    });

    bubbles.forEach(b => {
      b.y -= b.speed * dt;
      b.wobble += b.wobbleSpeed * dt;
      const wobbleX = Math.sin(b.wobble) * 6;
      if (b.y + b.r < 0) {
        b.y = seafloorY + Math.random() * 80;
        b.x = 50 + Math.random() * 900;
        b.r = 3 + Math.random() * 7;
        b.speed = 30 + (10 - b.r) * 6;
        b.wobble = Math.random() * Math.PI * 2;
        b.wobbleSpeed = 1 + Math.random();
      }
      b.el.setAttribute('cx', b.x + wobbleX);
      b.el.setAttribute('cy', b.y);
      b.el.setAttribute('r', b.r);
    });

    requestAnimationFrame(animate);
  }

  requestAnimationFrame(animate);
})();
</script>
