<script>
(function(){
  const svgNS = "http://www.w3.org/2000/svg";
  const fish1 = document.getElementById("fish1");
  const fish2 = document.getElementById("fish2");
  const bubbleLayer = document.getElementById("bubbles");
  const seaweed1 = document.getElementById("seaweed1");
  const seaweed2 = document.getElementById("seaweed2");
  const seaweed3 = document.getElementById("seaweed3");

  const fish = [
    { el: fish1, x: -100, y: 220, speed: 38, dir: 1 },
    { el: fish2, x: 900, y: 400, speed: 82, dir: -1 }
  ];

  const bubbles = [];
  const bubbleCount = 8;
  for (let i = 0; i < bubbleCount; i++) {
    const r = 3 + Math.random() * 8;
    const x = 50 + Math.random() * 900;
    const y = 540 + Math.random() * 40;
    const c = document.createElementNS(svgNS, "circle");
    c.setAttribute("cx", x);
    c.setAttribute("cy", y);
    c.setAttribute("r", r);
    c.setAttribute("fill", "url(#bubbleGrad)");
    c.setAttribute("stroke", "#e1f5fe");
    c.setAttribute("stroke-width", "1");
    bubbleLayer.appendChild(c);
    bubbles.push({
      el: c,
      x: x,
      y: y,
      r: r,
      speed: 30 + (12 - r) * 6
    });
  }

  const weeds = [
    { el: seaweed1, x: 120, y: 540, phase: 0, speed: 1.1 },
    { el: seaweed2, x: 350, y: 540, phase: 1.7, speed: 0.9 },
    { el: seaweed3, x: 760, y: 540, phase: 3.1, speed: 1.3 }
  ];

  let lastTime = 0;
  function animate(time) {
    if (!lastTime) lastTime = time;
    let dt = (time - lastTime) / 1000;
    lastTime = time;
    if (dt > 0.1) dt = 0.1;

    for (const f of fish) {
      f.x += f.speed * f.dir * dt;
      if (f.dir > 0 && f.x > 1150) f.x = -150;
      if (f.dir < 0 && f.x < -150) f.x = 1150;
      f.el.setAttribute("transform", `translate(${f.x},${f.y}) scale(${f.dir},1)`);
    }

    for (const b of bubbles) {
      b.y -= b.speed * dt;
      if (b.y < -b.r) {
        b.y = 540 + Math.random() * 40;
        b.x = 50 + Math.random() * 900;
        b.r = 3 + Math.random() * 8;
        b.speed = 30 + (12 - b.r) * 6;
        b.el.setAttribute("r", b.r);
      }
      b.el.setAttribute("cx", b.x);
      b.el.setAttribute("cy", b.y);
    }

    for (const w of weeds) {
      const angle = Math.sin(time * 0.001 * w.speed + w.phase) * 5;
      w.el.setAttribute("transform", `translate(${w.x},${w.y}) rotate(${angle})`);
    }

    requestAnimationFrame(animate);
  }
  requestAnimationFrame(animate);
})();
</script>
