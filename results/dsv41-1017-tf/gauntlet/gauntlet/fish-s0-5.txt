<script>
(function () {
  var NS = 'http://www.w3.org/2000/svg';
  var svg = document.getElementById('scene');
  var seafloorEl = document.getElementById('seafloor');
  var rocksEl = document.getElementById('rocks');
  var bubbleLayer = document.getElementById('bubbles');
  var weeds = Array.prototype.slice.call(document.querySelectorAll('.weed'));
  var f1el = document.getElementById('fish1');
  var f2el = document.getElementById('fish2');

  var W = 0, H = 0, BASE = 0;
  var placed = false;
  var bubbles = [];
  var last = 0, started = false;

  var f1 = { x: 0, dir: 1, drawn: 1, speed: 60, scale: 1, baseY: 0, phase: 0 };
  var f2 = { x: 0, dir: -1, drawn: -1, speed: 130, scale: 0.62, baseY: 0, phase: 1.7 };

  var weedSpec = [
    { fx: 0.09, s: 1.00 },
    { fx: 0.27, s: 0.78 },
    { fx: 0.52, s: 1.18 },
    { fx: 0.76, s: 0.90 },
    { fx: 0.93, s: 0.70 }
  ];

  function seafloorPath(w, h, base) {
    var d = 'M 0 ' + h + ' L 0 ' + base;
    var n = Math.max(4, Math.round(w / 170));
    for (var i = 0; i < n; i++) {
      var x1 = w * (i + 0.5) / n;
      var x2 = w * (i + 1) / n;
      var cy = base - 8 - Math.sin(i * 1.7) * 8 - (i % 2 ? 6 : 0);
      d += ' Q ' + x1.toFixed(1) + ' ' + cy.toFixed(1) + ' ' + x2.toFixed(1) + ' ' + base.toFixed(1);
    }
    d += ' L ' + w + ' ' + h + ' Z';
    return d;
  }

  function buildRocks() {
    while (rocksEl.firstChild) rocksEl.removeChild(rocksEl.firstChild);
    var colors = ['#b98d58', '#a97e4c', '#c9a06a', '#8f6a40'];
    for (var i = 0; i < 9; i++) {
      var e = document.createElementNS(NS, 'ellipse');
      var x = W * (0.03 + i * 0.11 + ((i % 3) - 1) * 0.012);
      var rx = W * 0.016 + (i % 4) * W * 0.006;
      var ry = rx * (0.45 + (i % 2) * 0.15);
      e.setAttribute('cx', x.toFixed(1));
      e.setAttribute('cy', (BASE + ry * 0.25).toFixed(1));
      e.setAttribute('rx', rx.toFixed(1));
      e.setAttribute('ry', ry.toFixed(1));
      e.setAttribute('fill', colors[i % colors.length]);
      e.setAttribute('opacity', '0.9');
      rocksEl.appendChild(e);
    }
  }

  function layout() {
    W = window.innerWidth || document.documentElement.clientWidth;
    H = window.innerHeight || document.documentElement.clientHeight;
    svg.setAttribute('viewBox', '0 0 ' + W + ' ' + H);
    BASE = H * 0.88;

    seafloorEl.setAttribute('d', seafloorPath(W, H, BASE));
    buildRocks();

    var k = Math.max(0.5, Math.min(1.5, Math.min(W, H) / 820));
    for (var i = 0; i < weeds.length; i++) {
      var sp = weedSpec[i % weedSpec.length];
      var s = sp.s * (H / 900);
      weeds[i].setAttribute(
        'transform',
        'translate(' + (W * sp.fx).toFixed(1) + ' ' + (BASE + 4).toFixed(1) + ') scale(' + s.toFixed(3) + ')'
      );
    }

    f1.scale = k;
    f2.scale = k * 0.62;
    f1.baseY = H * 0.33;
    f2.baseY = H * 0.63;
    f1.speed = W / 22;
    f2.speed = W / 9;

    if (!placed) {
      f1.x = W * 0.12;
      f2.x = W * 0.78;
      placed = true;
    }
    f1.x = Math.max(-300, Math.min(W + 300, f1.x));
    f2.x = Math.max(-300, Math.min(W + 300, f2.x));
  }

  function createBubbles(n) {
    var sizeK = Math.max(0.7, Math.min(1.4, Math.min(W, H) / 900));
    for (var i = 0; i < n; i++) {
      var r = (3 + Math.random() * 9) * sizeK;
      var g = document.createElementNS(NS, 'g');

      var c = document.createElementNS(NS, 'circle');
      c.setAttribute('r', r.toFixed(1));
      c.setAttribute('fill', 'rgba(190,230,255,0.30)');
      c.setAttribute('stroke', 'rgba(255,255,255,0.70)');
      c.setAttribute('stroke-width', '1.1');

      var hi = document.createElementNS(NS, 'circle');
      hi.setAttribute('cx', (-r * 0.34).toFixed(1));
      hi.setAttribute('cy', (-r * 0.34).toFixed(1));
      hi.setAttribute('r', Math.max(0.9, r * 0.27).toFixed(1));
      hi.setAttribute('fill', '#ffffff');
      hi.setAttribute('opacity', '0.75');

      g.appendChild(c);
      g.appendChild(hi);
      bubbleLayer.appendChild(g);

      bubbles.push({
        el: g,
        r: r,
        x: Math.random() * W,
        y: Math.random() * BASE,
        speed: 45 + 55 * (1 - (r - 3) / 9),
        phase: Math.random() * Math.PI * 2,
        drift: 6 + Math.random() * 10
      });
    }
  }

  function updateFish(t, dt) {
    var pair = [[f1, f1el], [f2, f2el]];
    for (var i = 0; i < pair.length; i++) {
      var f = pair[i][0], el = pair[i][1];
      f.x += f.dir * f.speed * dt;

      var margin = 140 * f.scale;
      if (f.dir > 0 && f.x > W + margin) f.x = -margin;
      if (f.dir < 0 && f.x < -margin) f.x = W + margin;

      var y = f.baseY + Math.sin(t * 1.1 + f.phase) * (H * 0.015);
      var flip = (f.dir === f.drawn) ? 1 : -1;
      el.setAttribute(
        'transform',
        'translate(' + f.x.toFixed(1) + ' ' + y.toFixed(1) + ') scale(' +
        (f.scale * flip).toFixed(3) + ' ' + f.scale.toFixed(3) + ')'
      );
    }
  }

  function updateBubbles(t, dt) {
    for (var i = 0; i < bubbles.length; i++) {
      var b = bubbles[i];
      b.y -= b.speed * dt;
      b.x += Math.cos(t * 1.3 + b.phase) * b.drift * dt;

      if (b.y + b.r < 0) {
        b.x = Math.random() * W;
        b.y = BASE + Math.random() * H * 0.03;
        b.phase = Math.random() * Math.PI * 2;
      }
      b.el.setAttribute('transform', 'translate(' + b.x.toFixed(1) + ' ' + b.y.toFixed(1) + ')');
    }
  }

  function frame(now) {
    if (!started) {
      started = true;
      last = now;
    }
    var dt = (now - last) / 1000;
    last = now;
    if (dt > 0.05) dt = 0.05;
    if (dt < 0) dt = 0;

    var t = now / 1000;
    updateFish(t, dt);
    updateBubbles(t, dt);

    requestAnimationFrame(frame);
  }

  window.addEventListener('resize', layout);

  layout();
  createBubbles(14);
  requestAnimationFrame(frame);
})();
</script>
